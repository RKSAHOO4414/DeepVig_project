# vig_ti_model.py
"""
Pyramid ViG-Ti model definition.
Adapted from the official ViG implementation:
https://github.com/huawei-noah/Efficient-AI-Backbones/tree/master/vig_pytorch
"""

import torch
import torch.nn as nn
import torch.nn.functional as F
from timm.models.layers import DropPath, trunc_normal_


class MRConv2d(nn.Module):
    """Max-Relative Graph Convolution (used in ViG)."""

    def __init__(self, in_channels, out_channels, act='relu', norm=None, bias=True):
        super().__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.act = nn.ReLU() if act == 'relu' else nn.GELU()

        self.fc = nn.Linear(in_channels, out_channels, bias=bias)
        self.norm = nn.BatchNorm2d(out_channels) if norm == 'bn' else nn.Identity()

    def forward(self, x, edge_index):
        # x: (N, C), edge_index: (2, E)
        row, col = edge_index
        x_j = x[row] - x[col]  # max-relative
        x_j = F.relu(x_j) if isinstance(self.act, nn.ReLU) else self.act(x_j)
        out = torch.zeros_like(x)
        out.index_add_(0, col, x_j)
        out = self.fc(out)
        return self.norm(out)


class Grapher(nn.Module):
    """
    Grapher module: graph construction (kNN) + graph convolution + FFN.
    This is the core building block of ViG.
    """

    def __init__(self, in_channels, kernel_size=9, dilation=1, drop_path=0.0):
        super().__init__()
        self.k = kernel_size  # k for kNN
        self.dilation = dilation
        self.channels = in_channels

        # Graph convolution (max-relative)
        self.graph_conv = MRConv2d(in_channels, in_channels * 2, act='relu', norm='bn')

        # FFN
        self.ffn = nn.Sequential(
            nn.Conv2d(in_channels * 2, in_channels * 4, 1, bias=False),
            nn.BatchNorm2d(in_channels * 4),
            nn.GELU(),
            nn.Conv2d(in_channels * 4, in_channels, 1, bias=False),
            nn.BatchNorm2d(in_channels),
        )

        self.drop_path = DropPath(drop_path) if drop_path > 0. else nn.Identity()

    def _build_knn_graph(self, x):
        """
        Build kNN graph from node features.
        x: (B, C, H, W) -> flatten to (B*H*W, C)
        Returns edge_index for each batch (concatenated).
        """
        B, C, H, W = x.shape
        N = H * W
        x_flat = x.permute(0, 2, 3, 1).reshape(B, N, C)  # (B, N, C)

        # Compute pairwise cosine similarity (batched)
        x_norm = F.normalize(x_flat, dim=-1)
        sim = torch.bmm(x_norm, x_norm.transpose(1, 2))  # (B, N, N)

        # Top-k neighbors (excluding self)
        sim.diagonal(dim1=1, dim2=2).fill_(-float('inf'))
        _, knn_idx = sim.topk(self.k, dim=-1)  # (B, N, k)

        # Build edge_index
        row = torch.arange(N, device=x.device).view(1, N, 1).expand(B, -1, self.k)
        edge_index = torch.stack([row.reshape(-1), knn_idx.reshape(-1)], dim=0)  # (2, B*N*k)
        return edge_index, B, N

    def forward(self, x):
        B, C, H, W = x.shape
        # --- Graph Construction ---
        edge_index, B, N = self._build_knn_graph(x)

        # --- Graph Convolution ---
        x_flat = x.permute(0, 2, 3, 1).reshape(B * N, C)
        x_conv = self.graph_conv(x_flat, edge_index)
        x_conv = x_conv.reshape(B, H, W, -1).permute(0, 3, 1, 2)

        # --- FFN ---
        x_ffn = self.ffn(x_conv)

        # --- Residual ---
        x = self.drop_path(x_ffn) + x
        return x


class PyramidViG(nn.Module):
    """
    Pyramid ViG-Ti (tiny) architecture.
    Stages: 4 stages with depths [2, 2, 6, 2] as in the original paper.
    """

    def __init__(self, num_classes=1000, in_chans=3, embed_dim=48,
                 depths=(2, 2, 6, 2), drop_path_rate=0.1):
        super().__init__()
        self.num_classes = num_classes
        self.depths = depths

        # Stem: patch embedding (16x16 conv, stride 16)
        self.stem = nn.Sequential(
            nn.Conv2d(in_chans, embed_dim, kernel_size=16, stride=16, bias=False),
            nn.BatchNorm2d(embed_dim),
            nn.GELU(),
        )

        # Stages
        self.stages = nn.ModuleList()
        dpr = [x.item() for x in torch.linspace(0, drop_path_rate, sum(depths))]
        cur = 0
        for i, depth in enumerate(depths):
            dim = embed_dim * (2 ** i)
            if i > 0:
                # Downsample between stages
                downsample = nn.Sequential(
                    nn.Conv2d(embed_dim * (2 ** (i - 1)), dim, kernel_size=2, stride=2, bias=False),
                    nn.BatchNorm2d(dim),
                )
            else:
                downsample = nn.Identity()

            stage = nn.Sequential(
                downsample,
                *[Grapher(dim, kernel_size=9, drop_path=dpr[cur + j])
                  for j in range(depth)]
            )
            self.stages.append(stage)
            cur += depth

        # Classification head
        self.norm = nn.BatchNorm2d(embed_dim * 8)
        self.head = nn.Linear(embed_dim * 8, num_classes)

        self.apply(self._init_weights)

    def _init_weights(self, m):
        if isinstance(m, nn.Linear):
            trunc_normal_(m.weight, std=.02)
            if m.bias is not None:
                nn.init.constant_(m.bias, 0)
        elif isinstance(m, nn.LayerNorm):
            nn.init.constant_(m.bias, 0)
            nn.init.constant_(m.weight, 1.0)

    def forward_features(self, x):
        x = self.stem(x)
        for stage in self.stages:
            x = stage(x)
        return x

    def forward(self, x):
        x = self.forward_features(x)
        x = F.adaptive_avg_pool2d(x, 1).flatten(1)
        x = self.norm(x.unsqueeze(-1).unsqueeze(-1)).squeeze(-1).squeeze(-1)
        x = self.head(x)
        return x


def pyramid_vig_ti(num_classes=1000):
    """Factory function for Pyramid ViG-Ti."""
    return PyramidViG(
        num_classes=num_classes,
        embed_dim=48,
        depths=(2, 2, 6, 2),
        drop_path_rate=0.1,
    )

# utils/data.py
"""Dataset loaders and transforms for ImageNet-100."""

import os
from torchvision import datasets, transforms
from timm.data import Mixup, create_transform


def build_imagenet100_loaders(data_dir, batch_size=128, num_workers=8):
    """
    Build train/val loaders for ImageNet-100.

    Expected directory structure:
        data_dir/
            train/
                n01440764/  ... (100 class folders)
            val/
                n01440764/  ... (100 class folders)
    """
    train_dir = os.path.join(data_dir, 'train')
    val_dir = os.path.join(data_dir, 'val')

    # DeiT-style training transforms
    train_transform = create_transform(
        input_size=224,
        is_training=True,
        color_jitter=0.4,
        auto_augment='rand-m9-mstd0.5-inc1',
        interpolation='bicubic',
        re_prob=0.25,
        re_mode='pixel',
        re_count=1,
    )

    val_transform = create_transform(
        input_size=224,
        is_training=False,
        interpolation='bicubic',
        crop_pct=0.95,
    )

    train_dataset = datasets.ImageFolder(train_dir, transform=train_transform)
    val_dataset = datasets.ImageFolder(val_dir, transform=val_transform)

    train_loader = torch.utils.data.DataLoader(
        train_dataset, batch_size=batch_size, shuffle=True,
        num_workers=num_workers, pin_memory=True, drop_last=True,
    )
    val_loader = torch.utils.data.DataLoader(
        val_dataset, batch_size=batch_size, shuffle=False,
        num_workers=num_workers, pin_memory=True,
    )

    return train_loader, val_loader, len(train_dataset.classes)

# train_imagenet100.py
"""Train Pyramid ViG-Ti on ImageNet-100 (Experiment 0.1)."""

import os
import time
import argparse
import torch
import torch.nn as nn
import torch.optim as optim
from torch.cuda.amp import GradScaler, autocast
from timm.scheduler import CosineLRScheduler
from timm.data import Mixup
from timm.loss import SoftTargetCrossEntropy

from vig_ti_model import pyramid_vig_ti
from utils.data import build_imagenet100_loaders
from utils.logger import WandbLogger


def train_one_epoch(model, loader, optimizer, scaler, criterion, mixup_fn, device):
    model.train()
    total_loss, correct, total = 0.0, 0, 0
    for images, targets in loader:
        images, targets = images.to(device), targets.to(device)
        if mixup_fn is not None:
            images, targets = mixup_fn(images, targets)

        optimizer.zero_grad()
        with autocast():
            outputs = model(images)
            loss = criterion(outputs, targets)

        scaler.scale(loss).backward()
        scaler.step(optimizer)
        scaler.update()

        total_loss += loss.item() * images.size(0)
        _, pred = outputs.max(1)
        if mixup_fn is None:
            correct += pred.eq(targets).sum().item()
        total += images.size(0)

    return total_loss / total, correct / total


@torch.no_grad()
def validate(model, loader, criterion, device):
    model.eval()
    total_loss, correct, total = 0.0, 0, 0
    for images, targets in loader:
        images, targets = images.to(device), targets.to(device)
        outputs = model(images)
        loss = criterion(outputs, targets)
        total_loss += loss.item() * images.size(0)
        _, pred = outputs.max(1)
        correct += pred.eq(targets).sum().item()
        total += images.size(0)
    return total_loss / total, correct / total


def main(args):
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    print(f'Using device: {device}')

    # Data
    train_loader, val_loader, num_classes = build_imagenet100_loaders(
        args.data_dir, batch_size=args.batch_size, num_workers=args.num_workers
    )
    print(f'ImageNet-100: {num_classes} classes, {len(train_loader.dataset)} train images')

    # Model
    model = pyramid_vig_ti(num_classes=num_classes).to(device)
    n_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
    print(f'Pyramid ViG-Ti: {n_params / 1e6:.2f}M parameters')

    # Mixup / CutMix
    mixup_fn = Mixup(
        mixup_alpha=0.8, cutmix_alpha=1.0, cutmix_minmax=None,
        prob=1.0, switch_prob=0.5, mode='batch',
        label_smoothing=0.1, num_classes=num_classes,
    )
    train_criterion = SoftTargetCrossEntropy()
    val_criterion = nn.CrossEntropyLoss()

    # Optimizer
    optimizer = optim.AdamW(model.parameters(), lr=args.lr, weight_decay=args.weight_decay)
    scaler = GradScaler()

    # LR scheduler
    scheduler = CosineLRScheduler(
        optimizer, t_initial=args.epochs, lr_min=1e-6,
        warmup_t=args.warmup_epochs, warmup_lr_init=1e-6,
    )

    # Logger
    logger = WandbLogger(project='DeepViG-Phase0', name='ViG-Ti-ImageNet100')

    best_acc = 0.0
    for epoch in range(args.epochs):
        t0 = time.time()
        train_loss, train_acc = train_one_epoch(
            model, train_loader, optimizer, scaler, train_criterion, mixup_fn, device
        )
        val_loss, val_acc = validate(model, val_loader, val_criterion, device)
        scheduler.step(epoch)

        logger.log({
            'epoch': epoch,
            'train_loss': train_loss, 'train_acc': train_acc,
            'val_loss': val_loss, 'val_acc': val_acc,
            'lr': optimizer.param_groups[0]['lr'],
            'time': time.time() - t0,
        })

        print(f'Epoch {epoch+1}/{args.epochs} | '
              f'Train Loss: {train_loss:.4f} Acc: {train_acc:.4f} | '
              f'Val Loss: {val_loss:.4f} Acc: {val_acc:.4f} | '
              f'Time: {time.time()-t0:.1f}s')

        if val_acc > best_acc:
            best_acc = val_acc
            torch.save(model.state_dict(), os.path.join(args.output_dir, 'best.pth'))
            print(f'  -> New best accuracy: {best_acc:.4f}')

    print(f'\nBaseline reproduction complete. Best validation accuracy: {best_acc:.4f}')
    logger.finish()


if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--data_dir', type=str, default='./data/imagenet100')
    parser.add_argument('--output_dir', type=str, default='./outputs/vig_ti_imagenet100')
    parser.add_argument('--epochs', type=int, default=100)
    parser.add_argument('--batch_size', type=int, default=128)
    parser.add_argument('--lr', type=float, default=2e-3)
    parser.add_argument('--weight_decay', type=float, default=0.05)
    parser.add_argument('--warmup_epochs', type=int, default=20)
    parser.add_argument('--num_workers', type=int, default=8)
    args = parser.parse_args()
    os.makedirs(args.output_dir, exist_ok=True)
    main(args)

# profile_vig.py
"""
Profile Pyramid ViG-Ti to measure latency breakdown:
  - Graph construction (kNN)
  - Graph convolution
  - FFN
  - Total forward pass
  - GPU memory usage

Uses torch.profiler with CUDA activities.
"""

import torch
import torch.nn as nn
from torch.profiler import profile, ProfilerActivity, record_function
from vig_ti_model import pyramid_vig_ti


def profile_model(model, input_size=(1, 3, 224, 224), device='cuda', warmup=10, runs=50):
    """
    Profile the model and return per-component timings.
    """
    model.eval().to(device)
    dummy = torch.randn(*input_size, device=device)

    # Warmup
    with torch.no_grad():
        for _ in range(warmup):
            _ = model(dummy)

    torch.cuda.synchronize()

    # --- Full forward pass profiling ---
    with profile(
        activities=[ProfilerActivity.CPU, ProfilerActivity.CUDA],
        record_shapes=True,
        profile_memory=True,
    ) as prof:
        with torch.no_grad():
            for _ in range(runs):
                _ = model(dummy)

    print('\n' + '=' * 80)
    print('PROFILING RESULTS: Pyramid ViG-Ti (224x224, batch=1)')
    print('=' * 80)

    # Print top-20 ops by CUDA time
    print('\n--- Top 20 CUDA operations ---')
    print(prof.key_averages().table(sort_by='cuda_time_total', row_limit=20))

    # --- Component-level profiling using record_function ---
    print('\n--- Component-level timing (manually instrumented) ---')
    # We instrument the Grapher module by monkey-patching forward
    from vig_ti_model import Grapher

    timings = {'knn': [], 'graph_conv': [], 'ffn': [], 'total': []}

    original_grapher_forward = Grapher.forward

    def instrumented_forward(self, x):
        torch.cuda.synchronize()
        t0 = torch.cuda.Event(enable_timing=True)
        t1 = torch.cuda.Event(enable_timing=True)
        t2 = torch.cuda.Event(enable_timing=True)
        t3 = torch.cuda.Event(enable_timing=True)
        t4 = torch.cuda.Event(enable_timing=True)

        t0.record()
        B, C, H, W = x.shape
        edge_index, B, N = self._build_knn_graph(x)
        t1.record()

        x_flat = x.permute(0, 2, 3, 1).reshape(B * N, C)
        x_conv = self.graph_conv(x_flat, edge_index)
        x_conv = x_conv.reshape(B, H, W, -1).permute(0, 3, 1, 2)
        t2.record()

        x_ffn = self.ffn(x_conv)
        t3.record()

        x = self.drop_path(x_ffn) + x
        t4.record()

        torch.cuda.synchronize()
        timings['knn'].append(t0.elapsed_time(t1))
        timings['graph_conv'].append(t1.elapsed_time(t2))
        timings['ffn'].append(t2.elapsed_time(t3))
        timings['total'].append(t0.elapsed_time(t4))
        return x

    Grapher.forward = instrumented_forward

    with torch.no_grad():
        for _ in range(runs):
            _ = model(dummy)
    torch.cuda.synchronize()

    Grapher.forward = original_grapher_forward

    # Report averages
    import numpy as np
    print('\n--- Average per-Grapher-layer timings (ms) ---')
    for k in ['knn', 'graph_conv', 'ffn', 'total']:
        arr = np.array(timings[k])
        print(f'  {k:12s}: {arr.mean():.3f} ms  (std: {arr.std():.3f})')

    # Memory
    torch.cuda.reset_peak_memory_stats()
    with torch.no_grad():
        _ = model(dummy)
    print(f'\nPeak GPU memory: {torch.cuda.max_memory_allocated() / 1e6:.1f} MB')

    return prof


def main():
    device = 'cuda' if torch.cuda.is_available() else 'cpu'
    print(f'Profiling on: {device}')

    model = pyramid_vig_ti(num_classes=1000)
    n_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
    print(f'Pyramid ViG-Ti: {n_params / 1e6:.2f}M parameters')

    profile_model(model, input_size=(1, 3, 224, 224), device=device)

    # Also profile with larger batch
    print('\n' + '=' * 80)
    print('PROFILING WITH BATCH SIZE 32')
    print('=' * 80)
    profile_model(model, input_size=(32, 3, 224, 224), device=device)


if __name__ == '__main__':
    main()

# utils/logger.py
"""Lightweight logger with wandb and CSV fallback."""

import os
import csv
import wandb


class WandbLogger:
    def __init__(self, project='DeepViG', name=None, use_wandb=True):
        self.use_wandb = use_wandb
        if use_wandb:
            try:
                wandb.init(project=project, name=name)
                self.wandb = wandb
            except Exception as e:
                print(f'wandb init failed: {e}. Falling back to CSV.')
                self.use_wandb = False

        self.csv_path = os.path.join('logs', f'{name or "run"}.csv')
        os.makedirs('logs', exist_ok=True)
        self._csv_header_written = False

    def log(self, metrics: dict):
        if self.use_wandb:
            wandb.log(metrics)

        # CSV fallback
        with open(self.csv_path, 'a', newline='') as f:
            writer = csv.DictWriter(f, fieldnames=metrics.keys())
            if not self._csv_header_written:
                writer.writeheader()
                self._csv_header_written = True
            writer.writerow(metrics)

    def finish(self):
        if self.use_wandb:
            wandb.finish()

# utils/checkpoint.py
"""Checkpoint save/load utilities."""

import os
import torch


def save_checkpoint(model, optimizer, epoch, path):
    torch.save({
        'epoch': epoch,
        'model_state_dict': model.state_dict(),
        'optimizer_state_dict': optimizer.state_dict(),
    }, path)
    print(f'Checkpoint saved: {path}')


def load_checkpoint(model, optimizer, path, device='cuda'):
    ckpt = torch.load(path, map_location=device)
    model.load_state_dict(ckpt['model_state_dict'])
    if optimizer is not None:
        optimizer.load_state_dict(ckpt['optimizer_state_dict'])
    print(f'Checkpoint loaded from epoch {ckpt["epoch"]}')
    return ckpt['epoch']
